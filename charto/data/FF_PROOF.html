import finfetch as ff

# 1 — no API key, no account, no config
t = ff.Ticker("RELIANCE")

# 2 — canonical schema, across four statements
f = ff.available_fields()
for k in ("pl", "bs", "cf", "ratios"):
    print(f"{k:8} {len(f[k]):>3} fields")
print(f"{'TOTAL':8} {sum(len(v) for v in f.values()):>3} fields")

# 3 — price: 7 fields, three decades
h = t.history(period="max", interval="1d")
yrs = (h.index[-1] - h.index[0]).days / 365.25
print(f"\nprice fields  {len(h.columns)}  {list(h.columns)}")
print(f"daily bars    {len(h):,}   {h.index[0].date()} -> {h.index[-1].date()}  ({yrs:.1f} yrs)")
print(f"points/company {len(h.columns) * len(h):,}")

import finfetch as ff

# measure daily-bar depth across the size spectrum: mega-cap -> 2025 listing
SAMPLE = ["RELIANCE", "TCS", "ITC", "HDFCBANK", "AMBER", "QUESS", "KSOLVES", "TRUALT"]

bars, cols = [], None
for s in SAMPLE:
    h = ff.Ticker(s).history(period="max", interval="1d")
    cols = len(h.columns)
    bars.append(len(h))
    print(f"{s:10} {len(h):>6,} bars   {h.index[0].date()} -> {h.index[-1].date()}")

mean = sum(bars) / len(bars)
COMPANIES = 3500
print(f"\nprice fields          {cols}")
print(f"mean bars / company   {mean:,.0f}   (min {min(bars):,}  max {max(bars):,})")
print(f"companies             {COMPANIES:,}")
print(f"\n{COMPANIES:,} x {cols} x {mean:,.0f} = {COMPANIES*cols*mean/1e6:.1f}M price points")
